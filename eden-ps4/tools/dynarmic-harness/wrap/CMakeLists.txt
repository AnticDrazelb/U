cmake_minimum_required(VERSION 3.20)
project(dynwrap C CXX ASM)
include(CMakeDependentOption)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ARCHITECTURE x86_64)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
# EDEN_SRC: another Eden tree (a git worktree) instead of ~/eden (04.17)
if (DEFINED ENV{EDEN_SRC})
    set(EDEN_SRC $ENV{EDEN_SRC})
else()
    set(EDEN_SRC $ENV{HOME}/eden)
endif()
include_directories(${EDEN_SRC}/src)
# xbyak and fmt from ~/eden-deps (fetch-deps.sh) when no installed package provides them
find_package(xbyak 7 CONFIG QUIET)
if (NOT TARGET xbyak::xbyak)
    add_library(xbyak::xbyak INTERFACE IMPORTED GLOBAL)
    target_include_directories(xbyak::xbyak INTERFACE $ENV{HOME}/eden-deps/xbyak)
endif()
find_package(fmt 8 CONFIG QUIET)
if (NOT TARGET fmt::fmt)
    add_library(fmt::fmt INTERFACE IMPORTED GLOBAL)
    target_include_directories(fmt::fmt INTERFACE $ENV{HOME}/eden-deps/fmt/include)
    target_compile_definitions(fmt::fmt INTERFACE FMT_HEADER_ONLY)
endif()
add_subdirectory(${EDEN_SRC}/src/dynarmic dynarmic)
